" atop word
function! KubunSearch(word)
    call fzf#vim#grep(
        \ 'rg --column --line-number --no-heading --color=always '
        \ . shellescape(a:word) . ' ~/master/kubun.txt',
        \ 1,
        \ fzf#vim#with_preview(),
        \ 0)
endfunction

nnoremap <leader>k :call KubunSearch(expand('<cword>'))<CR>

" visual mode
function! KubunSearchVisual()
    normal! gv"zy
    call KubunSearch(@z)
endfunction

xnoremap <leader>k :<C-u>call KubunSearchVisual()<CR>
